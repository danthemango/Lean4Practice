-- example (b : ℕ) (A : Finset ℕ): (∃ a ∈ A, b ∣ a) → b ∣ ∏ a ∈ A, a := by
intro h1
choose a ha using h1
obtain ⟨ h2, h3 ⟩ := ha
apply insert_erase at h2
rw [← h2]
rw [prod_insert]
-- rw [dvd_iff_exists_eq_mul_left]

rw [dvd_iff_exists_eq_mul_left] at h3
rw [dvd_iff_exists_eq_mul_left]
obtain ⟨d, hd⟩ := h3
use d * (∏ x ∈ A.erase a, x)
rw [mul_comm (d * ∏ x ∈ A.erase a, x) b]

rw [mul_comm b d]
